   221→        body: JSON.stringify({ repo, token, isPublic: cfgPublic, createIfMissing: cfgAutoCreate }),
   222→      });

   224→      if (!data.ok) {
   225→        setCfgLoading(false);
   226→        setCfgError(data.error ?? "Configuration failed");
   227→        return;
   228→      }

   230→      try {
   231→        const pushRes = await fetch("/api/self-enhance/push", { method: "POST" });
   232→        const pushData = await pushRes.json();

   234→          emitOutput([
   235→            `[CONFIG] Connected to github.com/${data.repo} as ${data.login}`,
   236→            `[CONFIG] First proof push: commit ${pushData.commitSha}`,
   237→          ]);
   238→          const fresh = await fetch("/api/self-enhance/push-target");
   239→          if (fresh.ok) setCfgTarget(await fresh.json());
   240→          setCfgToken("");
   241→          setCfgShowToken(false);
   242→          setCfgStatusMsg(`Pushed commit ${pushData.commitSha}. Connected ✓`);

   244→          setCfgError(`Connected, but initial push failed: ${pushData.error ?? "unknown"}`);
   245→        }
   246→      } catch {
   247→        setCfgError("Connected, but initial push failed (network error).");
   248→      }
   249→    } catch (e) {
   250→      setCfgError((e as Error).message);

   252→      setCfgLoading(false);

   503→  async function handleSubmit(e: React.FormEvent) {
   504→    e.preventDefault();
   505→    const cmd = input;
   506→    setInput("");
   507→    if (!cmd.trim()) return;
   508→
   509→    // Echo the command to the terminal
   510→    setTerminalLines((prev) => [...prev, { type: "command", text: cmd }]);
   511→
   512→    // Execute and show output
   513→    const output = await executeCommand(cmd);
   514→    if (output && output.length > 0) {
   515→      setTerminalLines((prev) => [
   516→        ...prev,
   517→        ...output.map((line) => ({ type: "output" as const, text: line })),
   518→      ]);
   519→    }
   520→
   521→    // Force a status poll
   522→    setTimeout(async () => {
   523→      try {
   524→        const res = await fetch("/api/self-enhance/status");
   525→        if (res.ok) setStatus(await res.json());
   526→      } catch {
   527→        /* ignore */
   528→      }
   529→    }, 500);
   530→  }

   532→  function handleKeyDown(e: React.KeyboardEvent) {
   533→    if (e.key === "ArrowUp") {
   534→      e.preventDefault();
   535→      if (history.length > 0) {
   536→        const newIdx = historyIdx === -1 ? history.length - 1 : Math.max(0, historyIdx - 1);
   537→        setHistoryIdx(newIdx);
   538→        setInput(history[newIdx]);
   539→      }
   540→    } else if (e.key === "ArrowDown") {
   541→      e.preventDefault();
   542→      if (historyIdx !== -1) {
   543→        const newIdx = historyIdx + 1;
   544→        if (newIdx >= history.length) {
   545→          setHistoryIdx(-1);
   546→          setInput("");
   547→        } else {
   548→          setHistoryIdx(newIdx);
   549→          setInput(history[newIdx]);
   550→        }
   551→      }
   552→    }
   553→  }
   554→
   555→  const logs = status?.logs ?? [];
   556→  const s = status?.stats;
   557→  const q = status?.queue;
   558→  const running = status?.running ?? false;
   559→

   561→    <div
   562→      style={{
   563→        position: "fixed",
   564→        inset: 0,
   565→        background: "#000",
