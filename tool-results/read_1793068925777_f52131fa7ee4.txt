     1→/**
     2→ * Self-Enhance Loop — the system enhances its OWN source code.
     3→ *
     4→ * Instead of fetching a remote repo, mutating, and pushing back, this loop:
     5→ *   1. Scans /home/z/my-project/src/lib/pipeline/ for .ts files (its own code)
     6→ *   2. For each file: reads it, LLM-refactors it, verifies with tsc --noEmit
     7→ *   3. If it compiles: writes the refactored code LIVE (hot-reload picks it up)
     8→ *   4. If it doesn't compile: reverts and records a postmortem
     9→ *   5. After each successful enhancement: git add + commit + push (the proof)
    10→ *
    11→ * The GitHub push contains the CURRENT live state of the running system.
    12→ * When you see a commit on GitHub, that's exactly what the server is running.
    13→ */

    26→/**
    27→ * Find test files that cover the given source file.
    28→ *
    29→ * Looks for `*.test.ts` files that either:
    30→ *   - Sit next to the source file (same name, .test.ts suffix)
    31→ *   - Import the source file by its relative path
    32→ *
    33→ * This is the test-gate infrastructure for Direction 4 (tests as the gate).
    34→ * Right now no test files exist in the project, so this returns [] and the
    35→ * test gate is a no-op. The infrastructure is ready for when tests are added.
    36→ */
    37→function findTestFilesFor(sourceFilePath: string): string[] {
    38→  const dir = dirname(sourceFilePath);
    39→  const base = basename(sourceFilePath, ".ts");
    40→  const testFiles: string[] = [];
    41→
    42→  // 1. Co-located test file: foo.ts -> foo.test.ts
    43→  const coLocated = join(dir, `${base}.test.ts`);
    44→  if (existsSync(coLocated)) testFiles.push(coLocated);
    45→
    46→  // 2. Scan the directory for .test.ts files that import the source file
    47→  try {
    48→    const entries = readdirSync(dir, { withFileTypes: true });
    49→    for (const entry of entries) {
    50→      if (!entry.isFile() || !entry.name.endsWith(".test.ts")) continue;
    51→      const testPath = join(dir, entry.name);
    52→      if (testPath === coLocated) continue; // already added
    53→      try {
    54→        const testContent = readFileSync(testPath, "utf-8");
    55→        // Check if it imports the source file by name
    56→        if (testContent.includes(`from "./${base}"`) || testContent.includes(`from './${base}'`)) {
    57→          testFiles.push(testPath);
    58→        }
    59→      } catch {
    60→        /* unreadable — skip */
    61→      }
    62→    }
    63→  } catch {
    64→    /* dir unreadable — skip */
    65→  }
    66→
    67→  return testFiles;
    68→}

   187→  /**
   188→   * Detect truncated LLM output. The LLM often hits its output token limit
   189→   * mid-comment on larger files, producing code with:
   190→   *   - Unbalanced braces (more opens than closes)
   191→   *   - Unterminated block comments (odd count of block-comment delimiters)
   192→   *   - Missing closing code fence at the end
   193→   * Catching this early avoids a ~10s wasted typecheck cycle.
   194→   */
   195→  private detectTruncation(code: string, original: string): { truncated: boolean; reason?: string } {
   196→    // 1. Unbalanced braces — count { and } (ignoring strings/comments is hard;
   197→    //    a rough count is enough to catch truncation since the original was balanced)
   198→    let opens = 0, closes = 0;
   199→    for (let i = 0; i < code.length; i++) {
   200→      if (code[i] === "{") opens++;
   201→      else if (code[i] === "}") closes++;
   202→    }
   203→    if (opens !== closes) {
   204→      return {
   205→        truncated: true,
   206→        reason: `unbalanced braces (${opens} { vs ${closes} })`,
   207→      };
   208→    }
   209→
   210→    // 2. Unterminated block comment — odd number of /* */ pairs
   211→    const blockCommentOpens = (code.match(/\/\*/g) || []).length;
   212→    const blockCommentCloses = (code.match(/\*\//g) || []).length;
   213→    if (blockCommentOpens !== blockCommentCloses) {
   214→      return {
   215→        truncated: true,
   216→        reason: `unterminated block comment (${blockCommentOpens} /* vs ${blockCommentCloses} */)`,
   217→      };
   218→    }
   219→
   220→    // 3. Refactored code is significantly shorter than original + missing closing
   221→    //    characters — likely cut off mid-generation
   222→    if (code.length < original.length * 0.5 && !code.endsWith("}") && !code.endsWith(";")) {
   223→      return {
   224→        truncated: true,
   225→        reason: `output too short (${code.length}b vs ${original.length}b original) and not properly terminated`,
   226→      };
   227→    }
   228→
   229→    return { truncated: false };
   230→  }

   232→  /**

   236→   * sees the refactored code until it passes verification.

   238→  private verifyRefactor(filePath: string, refactoredCode: string): { ok: boolean; errors: string[] } {
   239→    try {
   240→      const result = spawnSync(
   241→        "bun",

   243→        {

   247→          encoding: "utf-8",
   248→        }
   249→      );

   251→      const stdout = (result.stdout || "").trim();
   252→      if (!stdout) {
   253→        const stderr = (result.stderr || "").slice(0, 200);
   254→        return { ok: false, errors: [`Verifier produced no output${stderr ? `: ${stderr}` : ""}`] };
   255→      }

   257→      try {
   258→        const parsed = JSON.parse(stdout) as { ok: boolean; errors: string[] };
   259→        return parsed;
   260→      } catch {
   261→        return { ok: false, errors: [`Verifier output not JSON: ${stdout.slice(0, 200)}`] };
   262→      }
   263→    } catch (err) {
   264→      return { ok: false, errors: [`Verifier failed to run: ${(err as Error).message}`] };
   265→    }
   266→  }

   296→  /**
   297→   * Detect banner-style comments: decorative separator lines (====, ----, ****,
   298→   * ────, ════) or attribution/branding phrases. Plain JSDoc docstrings do NOT match.
   299→   */
   300→  private static looksLikeBanner(comment: string): boolean {
   301→    // A line that is essentially a separator run, optionally prefixed with "//"
   302→    const hasSeparatorLine = /(^|\n)\s*(?:\/\/)?\s*[=*\-#~─═]{4,}\s*(?=\n|$)/.test(comment);
   303→    // Attribution / branding phrases
   304→    const hasBranding =
   305→      /enhanced by|mutated by|refactored by|auto-generated|auto-refactor/i.test(comment);
   306→    return hasSeparatorLine || hasBranding;
   307→  }
   308→
   309→  /**
   310→   * Strip a LEADING banner comment (block or line-comment style) from refactored
   311→   * output. Conservative: only strips if it matches banner heuristics, so
   312→   * legitimate file docstrings survive.
   313→   */
   314→  private stripBannerComments(code: string): { code: string; stripped: boolean } {
   315→    let out = code;
   316→
   317→    // Leading block comment
   318→    const blockMatch = out.match(/^\/\*+\s*[\s\S]*?\*+\/\s*/);
   319→    if (blockMatch && SelfEnhanceLoop.looksLikeBanner(blockMatch[0])) {
   320→      out = out.slice(blockMatch[0].length);
   321→      return { code: out.replace(/^\s*\n/, ""), stripped: true };
   322→    }
   323→
   324→    // Leading run of // line comments
   325→    const lineMatch = out.match(/^(?:[ \t]*\/\/[^\n]*\n)+/);
   326→    if (lineMatch && SelfEnhanceLoop.looksLikeBanner(lineMatch[0])) {
   327→      out = out.slice(lineMatch[0].length);
   328→      return { code: out.replace(/^\s*\n/, ""), stripped: true };
   329→    }
   330→
   331→    return { code: out, stripped: false };
   332→  }

   334→  /**
   335→   * DEBATE GATE — a second LLM reviews the proposed refactor before write.
   336→   *
   337→   * The typecheck gate proves "compiles". The structural validator proves
   338→   * "didn't drop exports / wasn't truncated". NEITHER proves "the change is
   339→   * semantically correct and safe." A second LLM (Security + Architecture +
   340→   * Lead tribunal from debate-llm.ts) evaluates the candidate code and can
   341→   * REJECT or REQUIRES_REWRITE it. This is the missing semantic gate — the
   342→   * typecheck can't catch "added a null check that breaks the happy path."
   343→   *
   344→   * If the debate LLM is unavailable, fall back to APPROVED (don't block
   345→   * the whole loop on the second-LLM being down — the first gate already
   346→   * passed typecheck + structural).
   347→   */
   348→  private async debateGate(
   349→    filePath: string,
   350→    originalCode: string,
   351→    refactoredCode: string,
   352→    changeSummary: string
   353→  ): Promise<{ approved: boolean; reason: string; verdict?: string }> {
   354→    try {
   355→      const topic = `Proposed refactor of ${relative(PROJECT_ROOT, filePath)}: ${changeSummary.slice(0, 200)}`;
   356→      const result = await conductLlmDebate(topic, refactoredCode, filePath);
   357→      const consensus = result.verdict.consensus;
   358→
   359→      if (consensus === "APPROVED" || consensus === "CONDITIONAL_APPROVAL") {
   360→        return { approved: true, reason: `Debate: ${consensus}`, verdict: consensus };
   361→      }
   362→      // REJECTED or REQUIRES_REWRITE
   363→      return {
   364→        approved: false,
   365→        reason: `Debate rejected (${consensus}): ${result.verdict.leadSummary ?? "no rationale"}`,
   366→        verdict: consensus,
   367→      };
   368→    } catch (err) {
   369→      // Debate unavailable — don't block the loop, the first gate passed
   370→      return { approved: true, reason: `Debate unavailable (skipped): ${(err as Error).message.slice(0, 100)}` };
   371→    }
   372→  }

   374→  /**
   375→   * TEST GATE — run any test files that cover the target file.
   376→   *
   377→   * This is THE real signal from DIRECTIONS.md (Direction 4: tests as the
   378→   * gate). Typecheck proves "compiles". Tests prove "behavior preserved."
   379→   *
   380→   * Finds test files that import the target file and runs them with `bun test`.
   381→   * If any test fails, REJECT the refactor (don't write, don't push).
   382→   * If no tests exist for the file, the gate passes (no signal = no block).
   383→   *
   384→   * The refactored code must be written to disk for the test runner to see
   385→   * it — BUT we only write if typecheck already passed, and we REVERT
   386→   * immediately if tests fail. The window is small and the typecheck gate
   387→   * already guaranteed it compiles.
   388→   */
   389→  private testGate(filePath: string, refactoredCode: string): { passed: boolean; reason: string; testCount: number } {
   390→    const relPath = relative(PROJECT_ROOT, filePath);
   391→    const testFiles = findTestFilesFor(filePath);
   392→    if (testFiles.length === 0) {
   393→      return { passed: true, reason: "no tests for this file", testCount: 0 };
   394→    }
   395→
   396→    // Write the refactored code so the test runner sees it
   397→    const originalOnDisk = readFileSync(filePath, "utf-8");
   398→    writeFileSync(filePath, refactoredCode, "utf-8");
   399→
   400→    try {
   401→      let failures = 0;
   402→      let totalTests = 0;
   403→      for (const testFile of testFiles) {
   404→        const result = spawnSync("bun", ["test", testFile], {
   405→          cwd: PROJECT_ROOT,
   406→          timeout: 30000,
   407→          encoding: "utf-8",
   408→        });
   409→        totalTests++;
   410→        if (result.status !== 0) {
   411→          failures++;
   412→          const stderr = (result.stderr || "").slice(0, 300);
   413→          // Revert immediately so the broken code isn't live
   414→          writeFileSync(filePath, originalOnDisk, "utf-8");
   415→          return {
   416→            passed: false,
   417→            reason: `test failed: ${testFile} — ${stderr.slice(0, 200)}`,
   418→            testCount: totalTests,
   419→          };
   420→        }
   421→      }
   422→      return { passed: true, reason: `${testFiles.length} test file(s) passed`, testCount: totalTests };
   423→    } catch (err) {
   424→      // Revert on any unexpected error
   425→      writeFileSync(filePath, originalOnDisk, "utf-8");
   426→      return { passed: false, reason: `test runner error: ${(err as Error).message}`, testCount: 0 };
   427→    }
   428→  }

   430→  /**
   431→   * SATURATION CHECK — is this refactor actually meaningful, or just churn?
   432→   *
   433→   * Strips comments and whitespace from both original and refactored code,
   434→   * then compares. If the only changes are comments/whitespace/renamed
   435→   * locals, it's cosmetic churn — skip the write+push+debate cycle entirely.
   436→   * EMG's "SATURATION EQUILIBRIUM PROTOCOL": if the code is already optimal,
   437→   * output NO_CHANGES_NEEDED. We detect the equivalent after the fact.
   438→   */
   439→  private isCosmeticChange(original: string, refactored: string): boolean {
   440→    // Strip // comments, /* */ comments, and all whitespace, then compare
   441→    const strip = (s: string) =>
   442→      s
   443→        .replace(/\/\*[\s\S]*?\*\//g, "") // block comments
   444→        .replace(/\/\/[^\n]*/g, "") // line comments
   445→        .replace(/\s+/g, ""); // all whitespace
   446→    return strip(original) === strip(refactored);
   447→  }
   448→
   449→  /**
   450→   * ANTI-TAUTOLOGY CHECK — reject test files that assert nothing real.
   451→   *
   452→   * A test that always passes pins nothing — it's a fake pin. The ratchet
   453→   * is only as strong as its weakest test, so generated tests are checked
   454→   * for the classic tautology patterns before they're ever run:
   455→   *   - doesn't import the module under test (tests nothing)
   456→   *   - fewer than 2 test blocks (too thin to be a contract)
   457→   *   - fewer expect() calls than tests (a test without assertions)
   458→   *   - literal self-comparisons: expect(X).toBe(X)
   459→   *   - simple-expression self-comparisons: expect(f(1)).toBe(f(1))
   460→   * The debate gate adds semantic review on top of these mechanical checks.
   461→   */
   462→  private antiTautologyCheck(testCode: string, importBase: string): { ok: boolean; issues: string[] } {
   463→    const issues: string[] = [];
   464→
   465→    // Must import the module under test (relative import from the same dir)
   466→    if (!testCode.includes(`from "./${importBase}"`)) {
   467→      issues.push(`does not import the module under test (expected 'from "./${importBase}"')`);
   468→    }
   469→
   470→    // Need at least 2 test blocks to be a meaningful contract
   471→    const testBlocks = (testCode.match(/\btest\s*\(/g) ?? []).length;
   472→    if (testBlocks < 2) {
   473→      issues.push(`only ${testBlocks} test() block(s) — need at least 2`);
   474→    }
   475→
   476→    // Every test needs at least one assertion
   477→    const expects = (testCode.match(/\bexpect\s*\(/g) ?? []).length;
   478→    if (testBlocks > 0 && expects < testBlocks) {
   479→      issues.push(`${expects} expect() call(s) for ${testBlocks} test(s) — every test needs assertions`);
   480→    }
   481→
   482→    // Literal self-comparison: expect(true).toBe(true), expect(5).toBe(5), expect(x).toBe(x)
   483→    const selfCompare = /expect\s*\(\s*([^()]+?)\s*\)\s*\.\s*toBe\s*\(\s*([^()]+?)\s*\)/g;
   484→    let m: RegExpExecArray | null;
   485→    while ((m = selfCompare.exec(testCode)) !== null) {
   486→      if (m[1].trim() === m[2].trim()) {
   487→        issues.push(`tautological assertion: expect(${m[1].trim()}).toBe(${m[2].trim()}) — asserts nothing`);
   488→        break; // one report is enough
   489→      }
   490→    }
   491→
   492→    return { ok: issues.length === 0, issues };
   493→  }

   495→  /**
   496→   * THE RATCHET — write characterization tests for an untested module.
   497→   *
   498→   * This is Direction 4 made real: the system writes its own tests, the
   499→   * tests become the gate for every future refactor of that module. Each
   500→   * pass either adds pins (tightening the behavioral contract) or refactors
   501→   * within them. Behavior gets increasingly locked while implementation
   502→   * gets increasingly clean — and a refactor that breaks a pin is rejected
   503→   * by the test gate built in the previous step.
   504→   *
   505→   * Gates on the generated test file (in order):
   506→   *   1. truncation + anti-tautology (mechanical, instant)
   507→   *   2. WRITE TEMP → `bun test` — must PASS on the current code
   508→   *      (a test failing on current code pins wrong behavior — reject it,
   509→   *      do NOT "fix" the code to match a generated test)
   510→   *   3. `tsc --noEmit` — type errors in the test file → reject
   511→   *   4. debate gate (semantic review: is this a real behavioral test?)
   512→   *   5. keep + push. The pin is live.
   513→   *
   514→   * On any failure: DELETE the test file (it was new — nothing imports it,
   515→   * so the live system was never at risk) and record the attempt. After 2
   516→   * failed attempts per file, stop trying (don't burn LLM budget retrying).
   517→   *
   518→   * Test files are write-once: the refactor scan excludes *.test.ts, so the
   519→   * system can never weaken its own pins. The ratchet only tightens.
   520→   */
   521→  private async writeTestsForFile(filePath: string): Promise<boolean> {
   522→    const relPath = relative(PROJECT_ROOT, filePath);
   523→    const base = basename(filePath, ".ts");
   524→    const testPath = join(dirname(filePath), `${base}.test.ts`);
   525→
   526→    if (existsSync(testPath)) return false; // already pinned
   527→    const now = Date.now();
   528→    const rec = this.testWriteAttempts.get(relPath);
   529→    // Decay: forget failures older than 30 minutes — intermittent 429 windows
   530→    // shouldn't permanently skip a module
   531→    if (rec && now - rec.at > 30 * 60 * 1000) {
   532→      this.testWriteAttempts.delete(relPath);
   533→    } else if (rec && rec.attempts >= 2) {
   534→      return false; // known-failed recently — don't burn LLM budget retrying
   535→    }
   536→    this.testWriteAttempts.set(relPath, { attempts: (rec?.attempts ?? 0) + 1, at: now });
   537→
   538→    try {
   539→      const source = readFileSync(filePath, "utf-8");
   540→      if (source.length > 12000) return false; // too large for the LLM prompt
   541→
   542→      this.log(`Ratchet: writing characterization tests for ${relPath} (attempt ${(rec?.attempts ?? 0) + 1})...`, "info");
   543→
   544→      const userPrompt = `Module to pin: ${relPath}\n\n\`\`\`typescript\n${source}\n\`\`\`\n\nWrite ${base}.test.ts — a bun:test characterization suite pinning this module's current observable behavior. Import the module from "./${base}". Compute expected values yourself from the implementation. Output the summary first, then the test file.`;
   545→
   546→      const llmResult = await llmComplete({
   547→        systemPrompt: TEST_WRITER_SYSTEM,
   548→        userPrompt,
   549→        temperature: 0.2,
   550→      });
   551→      if (!llmResult.ok) {
   552→        this.log(`Ratchet: test-writing LLM unavailable for ${relPath}: ${(llmResult.error ?? "").slice(0, 120)}`, "warn");
   553→        return false;
   554→      }
   555→
   556→      const { summary, code } = extractSummaryAndCode(llmResult.content);
   557→      if (!code || !code.trim()) {
   558→        this.log(`Ratchet: LLM returned no test code for ${relPath}`, "warn");
   559→        return false;
   560→      }
   561→
   562→      // Gate 1: truncation + anti-tautology (mechanical, instant)
   563→      const truncation = this.detectTruncation(code, "");
   564→      if (truncation.truncated) {
   565→        this.log(`Ratchet: rejected test for ${relPath} (truncated: ${truncation.reason})`, "warn");
   566→        return false;
   567→      }
   568→      const taut = this.antiTautologyCheck(code, base);
   569→      if (!taut.ok) {
   570→        this.log(`Ratchet: rejected test for ${relPath} — ${taut.issues.join("; ")}`, "warn");
   571→        return false;
   572→      }
   573→
   574→      // Gate 2: write temp + run — the tests must PASS on the CURRENT code.
   575→      // A failing test pins wrong behavior (or found a real bug — but the
   576→      // system doesn't get to redefine behavior via its own generated tests,
   577→      // so conservative policy: reject the test, keep the code).
   578→      writeFileSync(testPath, code, "utf-8");
   579→      const run = spawnSync("bun", ["test", testPath], {
   580→        cwd: PROJECT_ROOT,
   581→        timeout: 60000,
   582→        encoding: "utf-8",
   583→      });
   584→      if (run.status !== 0) {
   585→        const stderr = ((run.stderr || "") + (run.stdout || "")).slice(0, 250);
   586→        rmSync(testPath);
   587→        this.log(`Ratchet: rejected test for ${relPath} (fails on current code — pins wrong behavior): ${stderr}`, "warn");
   588→        return false;
   589→      }
   590→
   591→      // Gate 3: typecheck — errors mentioning the test file → reject.
   592→      // (Pre-existing errors elsewhere are not the test's fault — bun test
   593→      // already proved the test runs; tsc-scoped errors are what matter.)
   594→      const tsc = spawnSync("bunx", ["tsc", "--noEmit"], {
   595→        cwd: PROJECT_ROOT,
   596→        timeout: 120000,
   597→        encoding: "utf-8",
   598→      });
   599→      const tscOut = `${tsc.stdout || ""}${tsc.stderr || ""}`;
   600→      if (tsc.status !== 0 && tscOut.includes(`${base}.test.ts`)) {
   601→        rmSync(testPath);
   602→        this.log(`Ratchet: rejected test for ${relPath} (typecheck errors in the test file)`, "warn");
   603→        return false;
   604→      }
   605→
   606→      // Gate 4: debate — semantic review of the test file
   607→      const debate = await this.debateGate(filePath, source, code, `characterization tests: ${summary.slice(0, 200)}`);
   608→      if (!debate.approved) {
   609→        rmSync(testPath);
   610→        this.log(`Ratchet: rejected test for ${relPath} by debate: ${debate.reason.slice(0, 150)}`, "warn");
   611→        return false;
   612→      }
   613→
   614→      // All gates passed — the pin is LIVE. Future refactors of this module
   615→      // now run against these tests (the test gate finds co-located tests).
   616→      this.stats.testsWritten++;
   617→      const testCount = (code.match(/\btest\s*\(/g) ?? []).length;
   618→      this.log(`Ratchet: PINNED ${relPath} with ${testCount} characterization tests — future refactors run against them (${debate.reason})`, "success");
   619→
   620→      const pushResult = await this.gitPush(`test: pin behavior of ${relPath}\n\n${summary}\n\n(characterization tests — the ratchet: refactors that break these are rejected)`);
   621→      if (pushResult.pushed) {
   622→        this.log(`Ratchet: pushed test pin ${pushResult.commitSha}`, "success");
   623→      }
   624→      return true;
   625→    } catch (err) {
   626→      // Clean up any partially-written test file — never leave a broken pin
   627→      try {
   628→        if (existsSync(testPath)) rmSync(testPath);
   629→      } catch {
   630→        /* ignore */
   631→      }
   632→      this.log(`Ratchet: error writing tests for ${relPath}: ${(err as Error).message}`, "error");
   633→      return false;
   634→    }
   635→  }
   636→
   637→  /**
   638→   * Enhance a single file: LLM refactor → verify tsc → write live → git push.
   639→   */
   640→  async enhanceFile(filePath: string): Promise<SelfEnhanceResult> {
   641→    const relPath = relative(PROJECT_ROOT, filePath);
   642→    this.currentFile = relPath;
   643→    const startTime = Date.now();
   644→
   645→    const result: SelfEnhanceResult = {
   646→      filePath: relPath,
   647→      enhanced: false,
   648→      reason: "",
   649→      oldBytes: 0,
   650→      newBytes: 0,
   651→      diffBytes: 0,
   652→      llmUsed: false,
   653→      llmLatencyMs: 0,
   654→      compiled: false,
   655→      pushed: false,
   656→    };
   657→
   658→    try {
   659→      // 1. Read the current file
   660→      const originalCode = readFileSync(filePath, "utf-8");
   661→      result.oldBytes = originalCode.length;
   662→
   663→      // 1a. Skip files that are too large for the LLM to refactor without truncating.
   664→      // The model's output token limit means files over ~12KB often get cut off
   665→      // mid-comment, producing syntax errors that waste a full typecheck cycle.
   666→      const MAX_FILE_BYTES = 12000;
   667→      if (originalCode.length > MAX_FILE_BYTES) {
   668→        result.reason = `Skipped (file too large: ${originalCode.length}b > ${MAX_FILE_BYTES}b)`;
   669→        this.log(`Skipped ${relPath} (${originalCode.length}b — too large for LLM, would truncate)`, "info");
   670→        return result;
   671→      }
   672→
   673→      // 1b. THE RATCHET — pin untested modules with characterization tests
   674→      // BEFORE refactoring. The pins become the gate for the refactor in
   675→      // this same pass (and all future ones): if the refactor below breaks
   676→      // a pinned behavior, the test gate rejects it. Each pass either adds
   677→      // pins or refactors within them. Test files are write-once (the scan
   678→      // excludes *.test.ts), so the system can never weaken its own pins.
   679→      // Protected files are skipped: they're the gate itself (importing the
   680→      // loop singleton in a test would have side effects).
   681→      if (!PROTECTED_FILES.has(relPath)) {
   682→        await this.writeTestsForFile(filePath);
   683→      }
   684→
   685→      // 2. Check for postmortem constraints
   686→      const postmortem = await postmortemStore.getConstraintsForFile(
   687→        "self",
   688→        "main",
   689→        relPath
   690→      );
   691→
   692→      // 3. LLM improvement — ask for REAL improvements, not cosmetic refactoring
   693→      this.log(`Improving ${relPath}...`, "info");
   694→      const refactorSystem = `You are a senior code reviewer improving a TypeScript file inside an autonomous self-improvement pipeline. Your changes will be typechecked and pushed to GitHub as the system's live state.
   695→
   696→Your job: find and fix REAL issues. Do NOT make cosmetic changes (type annotations, magic numbers, syntax modernization). Those are pointless. Instead, look for:
   697→
   698→1. BUGS: null/undefined crashes, missing error handling, unhandled edge cases, race conditions, off-by-one errors, resource leaks (unclosed handles)
   699→2. ERROR HANDLING: bare try/catch that swallows errors, missing error context, unhelpful error messages. Add actionable error messages with context.
   700→3. INPUT VALIDATION: functions that assume valid input and would crash or misbehave on bad input. Add guard clauses with clear errors.
   701→4. PERFORMANCE: obvious inefficiencies — O(n²) where O(n) works, repeated computations in loops, unnecessary allocations, synchronous I/O that could be async.
   702→5. SECURITY: injection risks, unsafe eval, hardcoded secrets, missing input sanitization.
   703→6. DEAD CODE: unused imports, unreachable branches, commented-out code.
   704→
   705→HARD RULES (violating any rule fails the pipeline):
   706→- Preserve EVERY top-level exported symbol (function, class, interface, type, const). Do not rename or remove them.
   707→- Do NOT add header comments, banners, or branding.
   708→- Do NOT add unused interfaces or types.
   709→- Do NOT change the public API signature of any export (parameter names, types, return type must stay the same).
   710→- Keep changes focused — don't rewrite the whole file. Only fix real issues.
   711→- If there are NO real issues to fix, output the original code unchanged.
   712→
   713→OUTPUT FORMAT (mandatory):
   714→- First, write a SHORT summary (1-5 bullet points) of what you changed and why. Start with "## Changes" on its own line.
   715→- Then output the complete refactored code in a single fenced \`\`\`typescript block.
   716→- Example:
   717→## Changes
   718→- Added null check in parseX() — could crash on undefined input
   719→- Replaced bare catch with contextual error message in fetchY()
   720→- Added input validation for limit parameter (negative values caused infinite loop)
   721→
   722→\`\`\`typescript
   723→// the full file with fixes applied
   724→\`\`\`
   725→${postmortem.hasActivePostmortem ? `\n${postmortem.promptBlock}\n` : ""}`;
   726→
   727→      const refactorUser = `File: ${relPath}\n\n\`\`\`typescript\n${originalCode}\n\`\`\`\n\nFind real bugs, error handling gaps, input validation issues, performance problems, security risks, or dead code in this file. Fix them. If nothing needs fixing, return the file unchanged. Output the summary first, then the code.`;
   728→
   729→      const llmResult = await llmComplete({
   730→        systemPrompt: refactorSystem,
   731→        userPrompt: refactorUser,
   732→        temperature: 0.2,
   733→      });
   734→
   735→      result.llmLatencyMs = llmResult.latencyMs;
   736→
   737→      if (!llmResult.ok) {
   738→        const isRateLimit = /429|Too many requests|rate limit/i.test(llmResult.error ?? "");
   739→        if (isRateLimit) {
   740→          // Circuit breaker: track consecutive 429s and back off the whole cycle
   741→          this.consecutiveRateLimits++;
   742→          const backoffMs = Math.min(60000, 15000 * this.consecutiveRateLimits);
   743→          result.reason = `LLM rate-limited (429) — pausing ${backoffMs / 1000}s`;
   744→          this.log(
   745→            `LLM rate-limited (429) for ${relPath} — pausing ${backoffMs / 1000}s (consecutive #${this.consecutiveRateLimits})`,
   746→            "warn"
   747→          );
   748→          // Pause before returning so the next file doesn't immediately hit the API
   749→          await new Promise((r) => setTimeout(r, backoffMs));
   750→        } else {
   751→          this.consecutiveRateLimits = 0;
   752→          result.reason = `LLM unavailable: ${llmResult.error}`;
   753→          this.log(`LLM unavailable for ${relPath}: ${llmResult.error}`, "warn");
   754→        }
   755→        return result;
   756→      }
   757→
   758→      // LLM succeeded — reset the circuit breaker
   759→      this.consecutiveRateLimits = 0;
   760→
   761→      // Extract the summary (what changed + why) AND the code from the response
   762→      const { summary: changeSummary, code: refactoredCode } = extractSummaryAndCode(llmResult.content);
   763→      if (!refactoredCode || refactoredCode.trim().length === 0) {
   764→        result.reason = "LLM returned no code";
   765→        this.log(`LLM returned no code for ${relPath}`, "warn");
   766→        return result;
   767→      }
   768→
   769→      // 3b. Mechanical guard: strip decorative banner comments the LLM
   770→      // sometimes prepends to the code (defense-in-depth)
   771→      const banner = this.stripBannerComments(refactoredCode);
   772→      const cleanCode = banner.code;
   773→      if (banner.stripped) {
   774→        this.log(`Stripped decorative banner header from ${relPath} (LLM ignored the no-header rule)`, "warn");
   775→      }
   776→
   777→      result.llmUsed = true;
   778→      result.newBytes = cleanCode.length;
   779→      result.diffBytes = Math.abs(cleanCode.length - originalCode.length);
   780→
   781→      // 3c. Detect truncated LLM output before the expensive typecheck.
   782→      const truncation = this.detectTruncation(cleanCode, originalCode);
   783→      if (truncation.truncated) {
   784→        result.reason = `Rejected (LLM output truncated: ${truncation.reason})`;
   785→        this.log(`Rejected ${relPath} (LLM output truncated: ${truncation.reason})`, "warn");
   786→        await postmortemStore.recordFailure({
   787→          repo: "self",
   788→          branch: "main",
   789→          filePath: relPath,
   790→          failureType: "COMPILE_ERROR",
   791→          symptom: `LLM output truncated: ${truncation.reason}`,
   792→          evidence: `Original: ${originalCode.length}b, Refactored: ${cleanCode.length}b. The LLM hit its output token limit before finishing.`,
   793→          constraints: `The previous refactor of ${relPath} was truncated by the LLM's output limit. Make smaller, more targeted changes — do not rewrite the whole file.`,
   794→        });
   795→        this.stats.totalReverted++;
   796→        return result;
   797→      }
   798→
   799→      // 4. Check if anything actually changed
   800→      if (cleanCode.trim() === originalCode.trim()) {
   801→        result.reason = "No issues found";
   802→        this.log(`No change for ${relPath} (LLM found no real issues to fix)`, "info");
   803→        return result;
   804→      }
   805→
   806→      // 4a. SATURATION CHECK — if the only changes are comments/whitespace, it's
   807→      // cosmetic churn. Skip the expensive typecheck + debate + test gates.
   808→      // EMG's saturation protocol: "if already optimal, output NO_CHANGES_NEEDED."
   809→      if (this.isCosmeticChange(originalCode, cleanCode)) {
   810→        result.reason = "Skipped (only cosmetic changes — comments/whitespace)";
   811→        this.log(`Skipped ${relPath} (cosmetic-only change — no real improvement)`, "info");
   812→        return result;
   813→      }
   814→
   815→      // 5. Verify the refactored code IN MEMORY — nothing is written to disk
   816→      // until it passes typecheck. This prevents Turbopack from ever seeing
   817→      // broken code (the root cause of the stale browser error).
   818→      this.log(`Verifying ${relPath} (in-memory typecheck)...`, "info");
   819→      const verification = this.verifyRefactor(filePath, cleanCode);
   820→      result.compiled = verification.ok;
   821→
   822→      if (!verification.ok) {
   823→        // Refactor failed typecheck — do NOT write to disk. Turbopack never sees it.
   824→        const errorSummary = verification.errors.slice(0, 3).join("; ");
   825→        this.log(`Rejected ${relPath} (typecheck failed): ${errorSummary}`, "error");
   826→
   827→        // Record postmortem so the next attempt gets warned
   828→        await postmortemStore.recordFailure({
   829→          repo: "self",
   830→          branch: "main",
   831→          filePath: relPath,
   832→          failureType: "COMPILE_ERROR",
   833→          symptom: `Refactored code failed in-memory typecheck: ${errorSummary}`,
   834→          evidence: `Original: ${originalCode.length} bytes, Refactored: ${cleanCode.length} bytes. Errors: ${verification.errors.join("\n")}`,
   835→          constraints: `The previous refactor of ${relPath} did not compile. Make smaller, safer changes. Preserve all type signatures exactly.`,
   836→        });
   837→
   838→        result.reason = "Rejected (typecheck failed — nothing written to disk)";
   839→        return result;
   840→      }
   841→
   842→      // 5b. DEBATE GATE — a second LLM (Security + Architecture + Lead tribunal)
   843→      // reviews the proposed refactor for semantic correctness. The typecheck
   844→      // can't catch "added a null check that breaks the happy path." The debate
   845→      // can. If REJECTED or REQUIRES_REWRITE, don't write.
   846→      this.log(`Debating ${relPath} (second-LLM semantic review)...`, "info");
   847→      const debate = await this.debateGate(filePath, originalCode, cleanCode, changeSummary);
   848→      if (!debate.approved) {
   849→        this.log(`Rejected ${relPath} by debate: ${debate.reason}`, "warn");
   850→        await postmortemStore.recordFailure({
   851→          repo: "self",
   852→          branch: "main",
   853→          filePath: relPath,
   854→          failureType: "REVIEW_REJECTED",
   855→          symptom: `Debate rejected the refactor: ${debate.reason}`,
   856→          evidence: `Change summary: ${changeSummary.slice(0, 500)}`,
   857→          constraints: `The previous refactor of ${relPath} was rejected by the debate tribunal (${debate.verdict}). Address the review feedback before retrying.`,
   858→        });
   859→        result.reason = `Rejected by debate (${debate.verdict})`;
   860→        this.stats.totalReverted++;
   861→        return result;
   862→      }
   863→      this.log(`Debate approved ${relPath}: ${debate.reason}`, "success");
   864→
   865→      // 5c. TEST GATE — run any test files that cover this source file.
   866→      // This is THE real signal (Direction 4): tests prove behavior was preserved.
   867→      // No tests exist yet, so this is a no-op gate — but the infrastructure is
   868→      // ready. When tests are added, a refactor that breaks a test is rejected.
   869→      const testResult = this.testGate(filePath, cleanCode);
   870→      if (!testResult.passed) {
   871→        this.log(`Rejected ${relPath} by test gate: ${testResult.reason}`, "warn");
   872→        await postmortemStore.recordFailure({
   873→          repo: "self",
   874→          branch: "main",
   875→          filePath: relPath,
   876→          failureType: "OTHER",
   877→          symptom: `Test gate failed: ${testResult.reason}`,
   878→          evidence: `${testResult.testCount} test file(s) checked. The refactor broke existing behavior.`,
   879→          constraints: `The previous refactor of ${relPath} broke existing tests. Do not change observable behavior.`,
   880→        });
   881→        result.reason = `Rejected by test gate: ${testResult.reason}`;
   882→        this.stats.totalReverted++;
   883→        return result;
   884→      }
   885→      if (testResult.testCount > 0) {
   886→        this.log(`Test gate passed for ${relPath}: ${testResult.reason}`, "success");
   887→      }
   888→
   889→      // 6. Typecheck passed — NOW it's safe to write the improved code to the live file.
   890→      // Turbopack will hot-reload it and see only valid code.
   891→      // (If the test gate already wrote it, this is a no-op.)
   892→      writeFileSync(filePath, cleanCode, "utf-8");
   893→
   894→      // 9. Enhancement succeeded — the file is now LIVE (hot-reload will pick it up)
   895→      result.enhanced = true;
   896→      result.reason = changeSummary ? "Enhanced and live" : "Enhanced and live (no summary)";
   897→      this.stats.totalEnhanced++;
   898→      this.stats.lastEnhancementAt = new Date().toISOString();
   899→
   900→      // Log the enhancement WITH the change summary so you can see WHAT was fixed
   901→      const shortSummary = changeSummary
   902→        ? changeSummary.split("\n").slice(0, 3).join(" | ").slice(0, 200)
   903→        : "no summary provided";
   904→      this.log(`Enhanced ${relPath} (${result.diffBytes}b diff): ${shortSummary}`, "success");
   905→
   906→      // 10. Git commit + push (the proof) — include the change summary in the message
   907→      const commitMessage = changeSummary
   908→        ? `Self-enhancement: ${relPath}\n\n${changeSummary}\n\n(tsc verified, ${result.diffBytes}b diff)`
   909→        : `Self-enhancement: ${relPath} (tsc verified, ${result.diffBytes}b diff)`;
   910→      const pushResult = await this.gitPush(commitMessage);
   911→      result.pushed = pushResult.pushed;
   912→      result.commitSha = pushResult.commitSha;
   913→      if (pushResult.pushed) {
   914→        this.stats.totalPushed++;
   915→      }
   916→
   917→      // 11. Resolve any postmortem for this file
   918→      await postmortemStore.markResolved("self", "main", relPath);
   919→
   920→      return result;
   921→    } catch (err) {
   922→      result.reason = `Error: ${(err as Error).message}`;
   923→      this.log(`Error enhancing ${relPath}: ${(err as Error).message}`, "error");
   924→      return result;
   925→    }
   926→  }

   928→  /**
   929→   * Run one cycle: scan files, enhance each one sequentially.
   930→   */
   931→  async runCycle(): Promise<{ filesScanned: number; filesEnhanced: number; results: SelfEnhanceResult[] }> {
   932→    if (this.running) {
   933→      // Don't log — the continuous runner polls runCycle and this is normal
   934→      return { filesScanned: 0, filesEnhanced: 0, results: [] };
   935→    }
   936→    this.running = true;
   937→    this.currentFile = null;
   938→
   939→    try {
   940→      const files = this.scanFiles();
   941→      // Report unique files seen, not a cumulative count per cycle
   942→      this.stats.totalScanned = files.length;
   943→      this.log(`Scanned ${files.length} files in src/lib/pipeline/`, "info");
   944→
   945→      const results: SelfEnhanceResult[] = [];
   946→      let enhanced = 0;
   947→
   948→      for (const file of files) {
   949→        const result = await this.enhanceFile(file);
   950→        results.push(result);
   951→        if (result.enhanced) enhanced++;
   952→        if (result.reason.startsWith("Rejected")) this.stats.totalReverted++;
   953→
   954→        // Brief pause between files to avoid hammering the LLM API
   955→        await new Promise((r) => setTimeout(r, 2000));
   956→      }
   957→
   958→      this.stats.cyclesCompleted++;
   959→      this.currentFile = null;
   960→      this.log(`Cycle complete: ${enhanced}/${files.length} files enhanced`, "success");
   961→      return { filesScanned: files.length, filesEnhanced: enhanced, results };
   962→    } finally {
   963→      this.running = false;
   964→    }
   965→  }
   966→
   967→  /**
   968→   * Run continuously: cycle after cycle until stopped.
   969→   */
   970→  async runContinuous(onUpdate?: () => void): Promise<void> {
   971→    this.continuous = true;
   972→    this.log("Starting continuous self-enhancement loop", "success");
   973→    while (this.continuous) {
   974→      await this.runCycle();
   975→      if (onUpdate) onUpdate();
   976→      if (!this.continuous) break;
   977→      // Wait between cycles
   978→      this.log("Waiting 5s before next cycle...", "info");
   979→      await new Promise((r) => setTimeout(r, 5000));
   980→    }
   981→    this.log("Continuous self-enhancement loop exited", "info");
   982→  }
   983→
   984→  stop() {
   985→    this.running = false;
   986→    this.continuous = false;
   987→    this.log("Self-enhance loop stopped", "warn");
   988→  }
