   172→  async function submitConfig() {
   173→    setCfgLoading(true);
   174→    setCfgError(null);
   175→    setCfgStatusMsg(null);
   176→    const token = cfgToken.trim();
   177→    const repo = cfgRepo.trim();
   178→    if (!token || !repo) {
   179→      setCfgLoading(false);
   180→      setCfgError("Both token and repo are required.");
   181→      return;
   182→    }
   183→    try {
   184→      const res = await fetch("/api/self-enhance/push-target", {
   185→        method: "POST",
   186→        headers: { "Content-Type": "application/json" },
   187→        body: JSON.stringify({ repo, token, isPublic: cfgPublic, createIfMissing: cfgAutoCreate }),
   188→      });
   189→      const data = await res.json();
   190→      if (!data.ok) {
   191→        setCfgLoading(false);
   192→        setCfgError(data.error ?? "Configuration failed");
   193→        return;
   194→      }
   195→      setCfgStatusMsg(data.created ? `Created github.com/${data.repo}. Pushing first proof...` : `Connected as ${data.login}. Pushing first proof...`);
   196→      try {
   197→        const pushRes = await fetch("/api/self-enhance/push", { method: "POST" });
   198→        const pushData = await pushRes.json();
   199→        if (pushData.pushed) {
   200→          emitOutput([
   201→            `[CONFIG] Connected to github.com/${data.repo} as ${data.login}`,
   202→            `[CONFIG] First proof push: commit ${pushData.commitSha}`,
   203→          ]);
   204→          const fresh = await fetch("/api/self-enhance/push-target");
   205→          if (fresh.ok) setCfgTarget(await fresh.json());
   206→          setCfgToken("");
   207→          setCfgShowToken(false);
   208→          setCfgStatusMsg(`Pushed commit ${pushData.commitSha}. Connected ✓`);
   209→        } else {
   210→          setCfgError(`Connected, but initial push failed: ${pushData.error ?? "unknown"}`);
   211→        }
   212→      } catch {
   213→        setCfgError("Connected, but initial push failed (network error).");
   214→      }
   215→    } catch (e) {
   216→      setCfgError((e as Error).message);
   217→    } finally {
   218→      setCfgLoading(false);
   219→    }
   220→  }

   244→  async function executeCommand(cmd: string) {
   245→    const trimmed = cmd.trim();
   246→    if (!trimmed) return;
   247→
   248→    setHistory((prev) => [...prev, trimmed]);
   249→    setHistoryIdx(-1);
   250→
   251→    const lower = trimmed.toLowerCase();
   252→    const parts = lower.split(/\s+/);

   254→
   255→    switch (command) {

   257→        return [

   259→          "  start       — Start continuous self-enhancement loop",
   260→          "  stop        — Stop the loop",
   261→          "  trigger     — Run one enhancement cycle",
   262→          "  status      — Show current status",
   263→          "  push        — Push current live state to GitHub",
   264→          "  config      — Open Git config panel",

   266→          "  files       — List files being enhanced",

   268→          "  help        — Show this help",
   269→          "",
   270→          "CHAT MODE:",
   271→          "  Type anything that's not a command and the system will",
   272→          "  analyze the codebase, propose a plan, and ask you to approve.",
   273→          "  Example: 'add a search input that filters the file list'",
   274→          "  Then type 'yes' to apply, or anything else to cancel.",
   275→          "",
   276→        ];
   277→
   278→      case "start": {
   279→        try {
   280→          const res = await fetch("/api/self-enhance/start", { method: "POST" });
   281→          const data = await res.json();
   282→          return [data.message || "Started"];
   283→        } catch (e) {
   284→          return [`Error: ${(e as Error).message}`];
   285→        }
   286→      }
   287→
   288→      case "stop": {
   289→        try {
   290→          await fetch("/api/self-enhance/stop", { method: "POST" });
   291→          return ["Self-enhancement loop stopped"];
   292→        } catch (e) {
   293→          return [`Error: ${(e as Error).message}`];
   294→        }
   295→      }
   296→
   297→      case "config":
   298→      case "cfg": {
   299→        openConfig();
   300→        return ["Opening Git config panel... (or click [CONFIG] in the header)"];
   301→      }
   302→
   303→      case "trigger":
   304→      case "run": {
   305→        return ["Running one cycle... (check logs for progress)"];
   306→      }
   307→
   308→      case "status": {
   309→        const s = status;
   310→        if (!s) return ["No status available yet"];
   311→        return [
   312→          `Running:       ${s.running ? "YES" : "NO"}`,
   313→          `Files scanned: ${s.stats.totalScanned}`,
   314→          `Files enhanced: ${s.stats.totalEnhanced}`,
   315→          `Files reverted: ${s.stats.totalReverted}`,
   316→          `Pushes to git:  ${s.stats.totalPushed}`,
   317→          `Last enhanced:  ${s.stats.lastEnhancementAt ?? "never"}`,
   318→          `Current file:   ${s.currentFile ?? "idle"}`,
   319→        ];
   320→      }
   321→
   322→      case "push": {
   323→        try {
   324→          const res = await fetch("/api/self-enhance/push", { method: "POST" });
   325→          const data = await res.json();
   326→          if (data.pushed) return [`Pushed to GitHub: commit ${data.commitSha}`];
   327→          if (data.committed) return [`Committed ${data.commitSha} but push failed (no remote?)`];
   328→          return ["No changes to push"];
   329→        } catch (e) {
   330→          return [`Error: ${(e as Error).message}`];
   331→        }
   332→      }
   333→
   334→      case "logs": {
   335→        const logs = status?.logs ?? [];
   336→        if (logs.length === 0) return ["No logs yet"];
   337→        return logs.slice(0, 20).map((l) => `[${l.timestamp}] ${l.message}`);
   338→      }
   339→
   340→      case "files":
   341→      case "list": {
   342→        return [
   343→          "Self-enhancement targets (src/lib/pipeline/):",
   344→          "  core/       — unified-pipeline, ast-diff-gate, siphon, binaryShield",
   345→          "  emg/        — sanitizer, debate, halt, emg-rag",

   347→          "  postmortem/ — postmortem-store",
   348→          "",
   349→          "Protected (not modified):",
   350→          "  autonomous/ — loop-runtime, self-enhance-loop, loop-event-bus",
   351→          "  llm/         — agent-llm, debate-llm",
   352→        ];
   353→      }
   354→
   355→      case "clear":
   356→      case "cls":
   357→        setStatus(null);
   358→        return [""];
   359→
   360→      case "exit":
   361→      case "quit":
   362→        return ["Cannot exit — the system runs continuously. Use 'stop' to halt enhancement."];
   363→
   364→      case "yes":
   365→      case "y":
   366→      case "approve": {
   367→        if (!pendingPlan) {
   368→          return ["No pending plan to approve. Type a goal first."];
   369→        }
   370→        setTaskLoading(true);
   371→        try {
   372→          const res = await fetch("/api/self-enhance/chat-task/apply", {
   373→            method: "POST",
   374→            headers: { "Content-Type": "application/json" },
   375→            body: JSON.stringify({ ...pendingPlan, ok: true }),
   376→          });
   377→          const data = await res.json();
   378→          setPendingPlan(null);
   379→          if (data.ok) {
   380→            const lines = [
   381→              "APPLIED:",
   382→              ...data.results.map(
   383→                (r: { path: string; applied: boolean; reason: string }) =>
   384→                  `  ${r.applied ? "✓" : "✗"} ${r.path} — ${r.reason}`
   385→              ),
   386→            ];
   387→            if (data.pushed) {
   388→              lines.push(`Pushed to GitHub: commit ${data.commitSha}`);
   389→            } else if (data.error) {
   390→              lines.push(`⚠ ${data.error}`);
   391→            }
   392→            return lines;
   393→          }
   394→          return [`Failed: ${data.error}`];
   395→        } catch (e) {
   396→          setPendingPlan(null);
   397→          return [`Error applying plan: ${(e as Error).message}`];
   398→        } finally {
   399→          setTaskLoading(false);
   400→        }
   401→      }
   402→
   403→      case "no":
   404→      case "n":
   405→      case "cancel": {
   406→        if (pendingPlan) {
   407→          setPendingPlan(null);
   408→          return ["Plan cancelled."];
   409→        }
   410→        return ["Nothing to cancel."];
   411→      }
   412→
   413→      case "about":
   414→        return [
   415→          "Architecture Helper Pipeline v4.3",
   416→          "Self-enhancing autonomous refactoring system",
   417→          "The system modifies its own source code in src/lib/pipeline/,",
   418→          "verifies with tsc --noEmit, and pushes to GitHub as proof.",
   419→          "What you see on GitHub IS the current live state of the server.",
   420→          "",
   421→          "CHAT MODE: type a goal in plain English to direct the system.",
   422→        ];
   423→
   424→      default: {
   425→        // Not a command — treat as a chat task goal
   426→        if (taskLoading) return ["Task in progress, please wait..."];
   427→
   428→        setTaskLoading(true);
   429→        try {
   430→          const res = await fetch("/api/self-enhance/chat-task/propose", {
   431→            method: "POST",
   432→            headers: { "Content-Type": "application/json" },
   433→            body: JSON.stringify({ goal: trimmed }),
   434→          });
   435→          const plan = await res.json();
   436→
   437→          if (!plan.ok) {
   438→            return [`Could not plan: ${plan.error ?? "unknown error"}`];
   439→          }
   440→
   441→          // Store the plan for "yes" approval
   442→          setPendingPlan({
   443→            userGoal: plan.userGoal,
   444→            understanding: plan.understanding,
   445→            files: plan.files,
   446→            summary: plan.summary,
   447→          });
   448→
   449→          const lines = [
   450→            `UNDERSTANDING: ${plan.understanding}`,
   451→            "",
   452→            "PROPOSED CHANGES:",
   453→            ...plan.files.map(
   454→              (f: { path: string; action: string; reason: string }) =>
   455→                `  ${f.action.toUpperCase()}: ${f.path} — ${f.reason}`
   456→            ),
   457→            "",
   458→            `SUMMARY: ${plan.summary}`,
   459→            "",
   460→            "Type 'yes' to apply and push, or 'no' to cancel.",
   461→          ];
   462→          return lines;
   463→        } catch (e) {
   464→          return [`Error: ${(e as Error).message}`];
   465→        } finally {
   466→          setTaskLoading(false);
   467→        }
   468→      }
   469→    }
   470→  }

   472→  async function handleSubmit(e: React.FormEvent) {
   473→    e.preventDefault();
   474→    const cmd = input;
   475→    setInput("");
   476→    if (!cmd.trim()) return;
   477→
   478→    // Echo the command to the terminal
   479→    setTerminalLines((prev) => [...prev, { type: "command", text: cmd }]);
   480→
   481→    // Execute and show output
   482→    const output = await executeCommand(cmd);
   483→    if (output && output.length > 0) {
   484→      setTerminalLines((prev) => [
   485→        ...prev,
   486→        ...output.map((line) => ({ type: "output" as const, text: line })),
   487→      ]);
   488→    }
   489→
   490→    // Force a status poll
   491→    setTimeout(async () => {
   492→      try {
   493→        const res = await fetch("/api/self-enhance/status");
   494→        if (res.ok) setStatus(await res.json());
   495→      } catch {
   496→        /* ignore */
   497→      }
   498→    }, 500);
   499→  }

   501→  function handleKeyDown(e: React.KeyboardEvent) {
   502→    if (e.key === "ArrowUp") {
   503→      e.preventDefault();
   504→      if (history.length > 0) {
   505→        const newIdx = historyIdx === -1 ? history.length - 1 : Math.max(0, historyIdx - 1);
   506→        setHistoryIdx(newIdx);
   507→        setInput(history[newIdx]);
   508→      }
   509→    } else if (e.key === "ArrowDown") {
   510→      e.preventDefault();
   511→      if (historyIdx !== -1) {
   512→        const newIdx = historyIdx + 1;
   513→        if (newIdx >= history.length) {
   514→          setHistoryIdx(-1);
   515→          setInput("");
   516→        } else {
   517→          setHistoryIdx(newIdx);
   518→          setInput(history[newIdx]);
   519→        }
   520→      }
   521→    }
   522→  }
   523→
   524→  const logs = status?.logs ?? [];
   525→  const s = status?.stats;
   526→  const running = status?.running ?? false;
   527→
   528→  return (
   529→    <div
   530→      style={{
   531→        position: "fixed",
   532→        inset: 0,
   533→        background: "#000",
   534→        color: PROMPT_COLOR,
   535→        fontFamily: "'Courier New', 'Lucida Console', monospace",
   536→        fontSize: "14px",
   537→        lineHeight: "1.4",
   538→        padding: "8px",
   539→        overflow: "hidden",

   549→          marginBottom: "8px",

   552→          alignItems: "center",
   553→          justifyContent: "space-between",
   554→          gap: "8px",
   555→        }}
   556→      >
   557→        <div style={{ display: "flex", alignItems: "center", flexWrap: "wrap", minWidth: 0 }}>
   558→          <span style={{ color: DIM_COLOR }}>┌─ </span>
   559→          <span style={{ color: PROMPT_COLOR, fontWeight: "bold" }}>
   560→            ARCHITECTURE HELPER PIPELINE v4.3 — SELF-ENHANCE MODE
   561→          </span>
   562→          <span style={{ color: DIM_COLOR, marginLeft: "16px" }}>─┐</span>
   563→          <span style={{ color: DIM_COLOR, marginLeft: "16px" }}>│</span>
   564→          <span style={{ marginLeft: "8px", color: running ? "#00ff00" : "#ff4444" }}>
   565→            [{running ? "ENHANCING" : "IDLE"}]

   575→            [GIT: {pushTarget?.configured ? "ON" : "OFF"}]

   577→          <button

   579→            onClick={(e) => {
   580→              e.stopPropagation();
   581→              openConfig();
   582→            }}
   583→            aria-label="Open Git push configuration"
   584→            style={{
   585→              background: "#000",
   586→              border: `1px solid ${DIM_COLOR}`,
   587→              color: PROMPT_COLOR,
   588→              fontFamily: "inherit",

   634→
   635→      {/* Prompt */}
   636→      <div
   637→        style={{
   638→          borderTop: `1px solid ${DIM_COLOR}`,
   639→          paddingTop: "4px",
   640→          marginTop: "4px",
   641→          flexShrink: 0,
   642→          display: "flex",

   692→              background: "#000",
   693→              border: `1px solid ${PROMPT_COLOR}`,
   694→              color: PROMPT_COLOR,
   695→              fontFamily: "inherit",

   697→              padding: "16px",

   718→                  autoComplete="off"
   719→                  spellCheck={false}

   721→                />
   722→                <button
   723→                  type="button"
   724→                  onClick={() => setCfgShowToken((v) => !v)}
   725→                  style={{ background: "#000", border: `1px solid ${DIM_COLOR}`, color: PROMPT_COLOR, fontFamily: "inherit", fontSize: "inherit", padding: "4px 10px", cursor: "pointer" }}

   787→                [ CONNECT &amp; PUSH ]

   791→                  type="button"
   792→                  onClick={disconnectTarget}
   793→                  disabled={cfgLoading}
   794→                  style={{ background: "#000", border: `1px solid ${DIM_COLOR}`, color: PROMPT_COLOR, fontFamily: "inherit", fontSize: "inherit", padding: "4px 10px", cursor: cfgLoading ? "wait" : "pointer", opacity: cfgLoading ? 0.5 : 1 }}
   795→                >

   799→              <button type="button" onClick={closeConfig} style={{ background: "#000", border: `1px solid ${DIM_COLOR}`, color: PROMPT_COLOR, fontFamily: "inherit", fontSize: "inherit", padding: "4px 10px", cursor: "pointer" }}>
   800→                [ CLOSE ]

   811→
   812→            <div style={{ color: DIM_COLOR, marginTop: "12px", fontSize: "12px", lineHeight: 1.3 }}>
   813→              Token verified with GitHub. Token stored locally only — never in git config,
   814→              URLs, or the pushed repo. Need a token? github.com/settings/tokens (classic PAT, repo scope).
   815→            </div>
   816→          </div>
   817→        </div>
   818→      )}
   819→    </div>
